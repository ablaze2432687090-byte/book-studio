/* Admin-only glass contract for all editor types. Never changes project data. */
(() => {
 const key='book-studio-panel-transparency',limit=v=>Math.min(80,Math.max(0,Number(v)||0));let amount=45;
 try{const saved=localStorage.getItem(key);if(saved!==null&&saved.trim()!==''&&Number.isFinite(Number(saved)))amount=limit(saved);}catch{}
 const roots=new Set(),controls=new Set();
 function set(value,persist=true){amount=limit(value);for(const root of roots){root.style.setProperty('--panel-opacity',String(1-amount/100));root.classList.toggle('opaque-panels',amount===0);}for(const control of controls){const input=control.querySelector('input'),output=control.querySelector('output'),reset=control.querySelector('button');input.value=amount;input.setAttribute('aria-valuetext',amount+'% 透明');output.textContent=amount+'%';reset.disabled=amount===45;}if(persist)try{localStorage.setItem(key,String(amount));}catch{}return amount;}
 function bind(control){if(controls.has(control))return;controls.add(control);control.querySelector('input').addEventListener('input',e=>set(e.target.value));control.querySelector('button').onclick=e=>{e.preventDefault();set(45);};set(amount,false);}
 function control(prefix='panelTransparency'){const node=document.createElement('div');node.className='editor-slider panel-transparency';node.setAttribute('role','group');node.setAttribute('aria-label','操作栏外观');node.innerHTML='<label for="'+prefix+'">栏位透明度</label><input id="'+prefix+'" type="range" min="0" max="80" step="1" value="45" aria-label="操作栏透明度"><output id="'+prefix+'Value" for="'+prefix+'">45%</output><button id="'+prefix+'Reset" type="button" title="恢复默认透明度" aria-label="恢复默认透明度">↺</button>';bind(node);return node;}
 function mount({root,panels=[],bars=[],control:existing}){root.classList.add('editor-glass-workspace');roots.add(root);for(const node of [...panels,...bars])node?.setAttribute('data-editor-glass','');if(existing)bind(existing);set(amount,false);return {set};}
 function attachGallery(wall){const workspace=document.querySelector('.gallery-workspace');if(!workspace)return;const stage=workspace.querySelector('.gallery-stage'),head=stage.querySelector('.gallery-stage-head'),bar=stage.querySelector('.gallery-view-tools'),footer=stage.querySelector('.gallery-navigation');mount({root:workspace,panels:[workspace.querySelector('.gallery-settings'),workspace.querySelector('.gallery-assets')],bars:[head,bar,footer]});head.insertBefore(control(),head.querySelector('#galleryFullscreen'));const slot=document.createElement('div');slot.className='gallery-canvas-slot';wall.root.before(slot);slot.append(wall.root);workspace.classList.add('gallery-glass-workspace');
  let scheduled=0;function sync(){cancelAnimationFrame(scheduled);scheduled=requestAnimationFrame(()=>{const desktop=matchMedia('(min-width:981px)').matches,full=document.fullscreenElement===stage;const left=desktop&&!full?workspace.querySelector('.gallery-settings').getBoundingClientRect().width:0,right=desktop&&!full?workspace.querySelector('.gallery-assets').getBoundingClientRect().width:0;wall.viewInsets=desktop?{left,right,top:head.offsetHeight+bar.offsetHeight,bottom:footer.offsetHeight}:null;wall.layout();});}
  const observer=new ResizeObserver(sync);for(const node of [workspace,head,bar,footer,slot])observer.observe(node);window.addEventListener('resize',sync);document.addEventListener('fullscreenchange',sync);sync();
 }
 globalThis.StudioEditorChrome={mount,attachGallery,set,control,bind,get value(){return amount;}};
})();
